+==================================================================================================+
|  SECTION 01  //  BASH STARTUP FILE ORDER  —  what gets read, and when                            |
+==================================================================================================+

    LOGIN SHELL  (ssh, tty login, or bash --login)
      1. /etc/profile
      2. /etc/profile.d/*.sh
      3. ~/.bash_profile   OR   ~/.bash_login   OR   ~/.profile  (first found wins)
      4. ~/.bash_logout    (only on exit)

    INTERACTIVE NON-LOGIN SHELL  (typical terminal emulator window)
      1. /etc/bash.bashrc  (distro-dependent)
      2. ~/.bashrc

    NON-INTERACTIVE NON-LOGIN  (scripts, cron, git hooks)
      1. $BASH_ENV  (if set — usually empty)
      2. no rc files by default

    KEY INSIGHT:
      ~/.bashrc usually sources ~/.bash_profile — but not always.
      Most "I don't know what's running" bugs come from this asymmetry.


+==================================================================================================+
|  SECTION 02  //  SEE WHAT YOUR SHELL IS RIGHT NOW                                                |
+==================================================================================================+

    # What shell am I?
    echo $SHELL
    echo $0
    ps -p $$ -o cmd=

    # Am I a login shell?
    shopt -q login_shell && echo "LOGIN shell" || echo "non-login shell"

    # Am I interactive?
    [[ $- == *i* ]] && echo "INTERACTIVE" || echo "non-interactive"

    # What is my BASH_ENV?
    echo "BASH_ENV=${BASH_ENV:-<unset>}"

    # What is my ENV (POSIX)?
    echo "ENV=${ENV:-<unset>}"


+==================================================================================================+
|  SECTION 03  //  LIST ALL FILES THAT WOULD BE SOURCED  —  static check                          |
+==================================================================================================+

    # Login shell chain
    ls -la /etc/profile
    ls -la /etc/profile.d/
    ls -la ~/.bash_profile ~/.bash_login ~/.profile 2>/dev/null

    # Interactive shell chain
    ls -la /etc/bash.bashrc /etc/bashrc 2>/dev/null
    ls -la ~/.bashrc

    # Non-interactive
    [[ -n "$BASH_ENV" ]] && ls -la "$BASH_ENV"

    # Show the actual content of each
    cat /etc/profile
    cat /etc/profile.d/*.sh
    cat ~/.bash_profile 2>/dev/null
    cat ~/.bashrc 2>/dev/null


+==================================================================================================+
|  SECTION 04  //  TRACE WHAT IS SOURCED IN REAL TIME  —  the gold method                          |
+==================================================================================================+

    # METHOD A — verbose trace (shows every command executed)
    bash -x -l -i -c "exit" 2>&1 | less

    # METHOD B — trace only source/. commands
    bash -x -l -i -c "exit" 2>&1 | grep -E "^\+\+? (source|\.) "

    # METHOD C — capture to file for review
    bash -x -l -i -c "exit" > ~/bash_trace.log 2>&1
    less ~/bash_trace.log

    # METHOD D — inside a running shell, turn on tracing
    set -x
    source ~/.bashrc
    set +x


+==================================================================================================+
|  SECTION 05  //  INTERCEPT EVERY SOURCE CALL  —  highest visibility                              |
+==================================================================================================+

    # Override `source` and `.` to log every call
    source() {
        echo "[SOURCE] $(date +%T) -> $*" >> ~/.bash_source_log
        builtin source "$@"
    }

    # Then open a NEW terminal — it will log everything
    # Check the log:
    cat ~/.bash_source_log

    # Remove override when done:
    unset -f source


+==================================================================================================+
|  SECTION 06  //  TRACE WITH STRACE  —  kernel-level view                                         |
+==================================================================================================+

    # Every file bash opens (system call level)
    strace -f -e trace=openat,open bash -l -i -c "exit" 2>&1 \
      | grep -E "\.sh|\.bashrc|\.profile|\.bash_profile" \
      | grep -v ENOENT

    # Or with sudo (if bash is on a read-only fs)
    sudo strace -f -e trace=openat bash -l -i -c "exit" 2>&1 \
      | grep -oP '(?<=openat\(AT_FDCWD, ")[^"]+' \
      | sort -u


+==================================================================================================+
|  SECTION 07  //  FIND ALL FUNCTIONS, ALIASES, VARIABLES                                          |
+==================================================================================================+

    # All functions currently defined
    declare -F

    # Full function bodies
    declare -f

    # Just the names, sorted
    declare -F | awk '{print $3}' | sort

    # All aliases
    alias

    # All exported variables
    export -p

    # All shell variables (huge list)
    declare -p | less

    # Where did a function come from?
    type -a my_function_name
    declare -f my_function_name

    # Where did an alias come from?
    type -a my_alias_name


+==================================================================================================+
|  SECTION 08  //  FIND SCRIPTS THAT SOURCE OTHERS  —  grep the chain                              |
+==================================================================================================+

    # Which files in home reference `source` or `.`
    grep -rnE "^\s*(source|\.)\s+" \
      ~/.bashrc ~/.bash_profile ~/.profile ~/.bash_aliases 2>/dev/null

    # Check /etc/profile.d for suspicious additions
    grep -rnE "^\s*(source|\.)\s+" /etc/profile /etc/profile.d/ 2>/dev/null

    # Look for recent changes (last 30 days)
    find ~ -maxdepth 2 -name ".*rc" -o -name ".bash*" -o -name ".profile" 2>/dev/null \
      | xargs ls -la 2>/dev/null

    # Watch for new lines added recently
    ls -la ~/.bashrc ~/.bash_profile ~/.profile 2>/dev/null


+==================================================================================================+
|  SECTION 09  //  AUDIT FOR SUSPICIOUS SOURCES  —  safety check                                   |
+==================================================================================================+

    # Any source calls that pull from tmp, downloads, curl, wget, git?
    grep -rnE "(curl|wget|nc |/tmp/|/var/tmp/)" \
      ~/.bashrc ~/.bash_profile ~/.profile /etc/profile.d/ 2>/dev/null

    # Any eval or backticks in rc files? (common injection vector)
    grep -nE "eval|\\\$\\(|\\\`" ~/.bashrc ~/.bash_profile ~/.profile 2>/dev/null

    # Any unexpected background processes started?
    grep -nE "&\\s*$|nohup" ~/.bashrc ~/.bash_profile ~/.profile 2>/dev/null

    # Bash history of what you've recently typed
    tail -50 ~/.bash_history


+==================================================================================================+
|  SECTION 10  //  CHECK FOR HIDDEN HOOKS  —  less obvious entry points                            |
+==================================================================================================+

    # System-wide bash startup (Bazzite has these):
    ls -la /etc/bashrc /etc/bash.bashrc /etc/profile.d/*.sh 2>/dev/null

    # Any DEBUG trap set?
    trap -p DEBUG

    # Any PROMPT_COMMAND running behind your back?
    echo "PROMPT_COMMAND=$PROMPT_COMMAND"

    # Any startup scripts in /etc/X11 or KDE autostart?
    ls -la ~/.config/autostart/ 2>/dev/null

    # Any systemd user services?
    systemctl --user list-units --type=service --state=running
    systemctl --user list-timers --all


+==================================================================================================+
|  SECTION 11  //  FULL DIAGNOSTIC  —  copy/paste this whole block                                 |
+==================================================================================================+

    {
      echo "=== SHELL INFO ==="
      echo "SHELL=$SHELL"
      echo "BASH_ENV=${BASH_ENV:-<unset>}"
      echo "PROMPT_COMMAND=${PROMPT_COMMAND:-<unset>}"
      shopt -q login_shell && echo "LOGIN shell" || echo "non-login"
      [[ $- == *i* ]] && echo "INTERACTIVE" || echo "non-interactive"

      echo ""
      echo "=== FILES THAT EXIST ==="
      for f in /etc/profile /etc/bashrc /etc/bash.bashrc \
               ~/.bash_profile ~/.bash_login ~/.profile ~/.bashrc; do
        [[ -f "$f" ]] && echo "  present: $f" || echo "  absent : $f"
      done

      echo ""
      echo "=== PROFILE.D ==="
      ls /etc/profile.d/*.sh 2>/dev/null

      echo ""
      echo "=== SOURCE CALLS IN RC FILES ==="
      grep -HnE "^\s*(source|\.)\s+" \
        ~/.bashrc ~/.bash_profile ~/.profile /etc/profile /etc/profile.d/*.sh 2>/dev/null

      echo ""
      echo "=== ALIASES ==="
      alias

      echo ""
      echo "=== FUNCTION COUNT ==="
      echo "total: $(declare -F | wc -l)"

    } > ~/bash_diagnostic.txt 2>&1

    less ~/bash_diagnostic.txt

+==================================================================================================+
|  SECTION 13  //  IF SOMETHING IS RUNNING YOU DIDN'T ASK FOR                                      |
+==================================================================================================+

    # 1. Find the line
    grep -nE "topgrade|ujust|overlay|curl|wget|eval" ~/.bashrc ~/.bash_profile 2>/dev/null

    # 2. Comment it out
    nano ~/.bashrc
    # add # at start of the line

    # 3. Reload
    source ~/.bashrc

    # 4. Verify it's gone
    grep -nE "topgrade|ujust|overlay" ~/.bashrc

    # 5. If it's in /etc/profile.d/ (system-wide)
    ls -la /etc/profile.d/
    # identify the file, then:
    sudo nano /etc/profile.d/suspicious-file.sh
    # OR disable it:
    sudo mv /etc/profile.d/suspicious-file.sh /etc/profile.d/suspicious-file.sh.disabled


+==================================================================================================+
|  SECTION 14  //  THE ONE-LINE MENTAL MODEL                                                       |
+==================================================================================================+

    LOGIN     : /etc/profile -> /etc/profile.d/* -> ~/.bash_profile
    INTERACTIVE: /etc/bashrc -> ~/.bashrc
    NON-INTER:  $BASH_ENV only

    TRACE     : bash -x -l -i -c "exit" 2>&1
    INTERCEPT : override `source` function
    AUDIT     : grep -rnE "source|\\." ~/.bash* /etc/profile*
    KERNEL    : strace -f -e openat bash -l -i -c "exit"
