+==================================================================================================+
|  SECTION 01  //  THE THREE COMMANDS  —  all you actually need                                    |
+==================================================================================================+

    # START  (once per boot)
    ~/projects/cheatsheets/scripts/start-avalhla.sh

    # TALK  (anytime the server is up)
    ollama run avalhla

    # STOP  (frees VRAM, model stays built)
    pkill -f "ollama serve"


+==================================================================================================+
|  SECTION 02  //  AFTER REBOOT  —  one-time setup per session                                     |
+==================================================================================================+

    # Option A — manual block
    pkill -f "ollama serve" 2>/dev/null
    OLLAMA_CONTEXT_LENGTH=16384 OLLAMA_NUM_GPU=999 nohup ollama serve > ~/.ollama/serve.log 2>&1 &
    until curl -fsS http://127.0.0.1:11434/api/tags >/dev/null 2>&1; do sleep 1; done
    echo "Ollama is ready"

    # Option B — one-liner script
    ~/projects/cheatsheets/scripts/start-avalhla.sh


+==================================================================================================+
|  SECTION 03  //  TALK TO HER  —  chat modes                                                       |
+==================================================================================================+

    ollama run avalhla                          # interactive chat
    ollama run avalhla "one-shot question"      # single question

    # Inside the >>> prompt:
    /bye            exit cleanly (server keeps running)
    Ctrl+D          same as /bye
    Ctrl+C          force quit if stuck
    /?              help
    /clear          wipe current chat context
    /show info      current model details


+==================================================================================================+
|  SECTION 04  //  EXIT + STOP  —  how to leave the room                                            |
+==================================================================================================+

    # 1. Inside the >>> prompt
    /bye

    # 2. Back at shell — optional, frees ~5.5 GB VRAM
    pkill -f "ollama serve"

    # Verify server stopped (should say "could not connect")
    curl -fsS http://127.0.0.1:11434/api/tags


+==================================================================================================+
|  SECTION 05  //  EDIT + REBUILD  —  only after touching persona/avalhla.Modelfile                 |
+==================================================================================================+

    nano ~/projects/cheatsheets/persona/avalhla.Modelfile

    # Rebuild (server must be running)
    cd ~/projects/cheatsheets
    ollama create avalhla -f persona/avalhla.Modelfile

    # Relaunch
    ollama run avalhla


+==================================================================================================+
|  SECTION 06  //  INSPECT + VERIFY  —  make sure she is who she says she is                        |
+==================================================================================================+

    ollama list | grep avalhla                  # does she exist?
    ollama show avalhla --modelfile             # full build
    ollama show avalhla --parameters            # all params
    ollama show avalhla --system                # just SYSTEM block
    ollama show avalhla --template              # chat template
    ollama show avalhla --license               # base model license

    # Expected parameters:
    #   temperature     0.3
    #   top_p           0.9
    #   top_k           40
    #   repeat_penalty  1.05
    #   num_predict     2048
    #   num_ctx         16384


+==================================================================================================+
|  SECTION 07  //  SYSTEM UPDATES  —  YOU run these. Never topgrade.                                |
+==================================================================================================+

    ujust update                # Bazzite one-shot update
    rpm-ostree status           # current + previous deployment
    sudo rpm-ostree rollback    # boot previous working
    systemctl reboot            # apply pending deployment

    # Rules:
    #   topgrade must NOT run ujust   (check ~/.config/topgrade.toml)
    #   fstrim runs on its own timer  (never inside topgrade)
    #   never sudo dnf install on host


+==================================================================================================+
|  SECTION 08  //  TROUBLESHOOT  —  one fix per symptom                                             |
+==================================================================================================+

    SYMPTOM: "could not connect to ollama server"
    FIX:     Start it. Not an error.
             OLLAMA_CONTEXT_LENGTH=16384 OLLAMA_NUM_GPU=999 ollama serve &

    SYMPTOM: Forgot context / wrong hardware
    FIX:     Server not started with correct env vars.
             pkill -f "ollama serve"
             OLLAMA_CONTEXT_LENGTH=16384 OLLAMA_NUM_GPU=999 ollama serve &

    SYMPTOM: Out of memory / GPU spillover
    FIX:     Close Firefox, OBS, games. Or lower num_ctx to 8192 and rebuild.

    SYMPTOM: Wrong personality / wrong prompt
    FIX:     cd ~/projects/cheatsheets
             ollama create avalhla -f persona/avalhla.Modelfile

    SYMPTOM: Nuclear reset (model corrupted)
    FIX:     ollama rm avalhla
             ollama create avalhla -f persona/avalhla.Modelfile

    SYMPTOM: Topgrade running things it shouldn't
    FIX:     nano ~/.config/topgrade.toml
             Remove: "Bazzite Native Update" = "ujust update"
             Remove: "Trim SSD Cache" = "sudo fstrim -v /"
