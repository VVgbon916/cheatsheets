+==================================================================================================+
|  SECTION 01  //  THE THREE COMMANDS  —  all you actually need                                    |
+==================================================================================================+

    # START  (once per boot)
    ~/projects/cheatsheets/scripts/start-avalhla.sh

    # TALK  (anytime the container is up)
    ollama run avalhla

    # STOP  (frees VRAM, model stays built)
    systemctl --user stop ollama.service


+==================================================================================================+
|  SECTION 02  //  AFTER REBOOT  —  one command starts the container                              |
+==================================================================================================+

    systemctl --user start ollama.service
    sleep 3
    curl -fsS http://127.0.0.1:11434/api/tags | jq -r '.models[].name'

    # Or the wrapper script (does both)
    ~/projects/cheatsheets/scripts/start-avalhla.sh


+==================================================================================================+
|  SECTION 03  //  TALK TO HER  —  chat modes                                                       |
+==================================================================================================+

    ollama run avalhla                          # interactive chat
    ollama run avalhla "one-shot question"      # single question

    # Inside the >>> prompt:
    /bye            exit cleanly (container keeps running)
    Ctrl+D          same as /bye
    Ctrl+C          force quit if stuck
    /?              help
    /clear          wipe current chat context
    /show info      current model details


+==================================================================================================+
|  SECTION 04  //  MEMORY TOOLS  —  persistent across sessions                                     |
+==================================================================================================+

    ai-chat                 persistent chat, saves to ~/.ai-memory/
    ai-learn <path>         index files into her knowledge base
    ai-remember             she reflects on recent sessions
    ai-ask "question"       one-shot with recent context
    ai-ask -n 10 "..."      load last 10 turns from today


+==================================================================================================+
|  SECTION 05  //  EDIT + REBUILD  —  only after touching persona/avalhla.Modelfile               |
+==================================================================================================+

    nano ~/projects/cheatsheets/persona/avalhla.Modelfile

    cd ~/projects/cheatsheets
    ollama create avalhla -f persona/avalhla.Modelfile

    ollama run avalhla


+==================================================================================================+
|  SECTION 06  //  INSPECT + VERIFY  —  make sure she is who she says she is                       |
+==================================================================================================+

    ollama list | grep avalhla                  # does she exist?
    ollama show avalhla --modelfile             # full build
    ollama show avalhla --parameters            # all params
    ollama show avalhla --system                # just SYSTEM block

    # Expected parameters:
    #   temperature     0.3
    #   top_p           0.9
    #   top_k           40
    #   repeat_penalty  1.05
    #   num_predict     2048
    #   num_ctx         16384

    systemctl --user status podman-auto-update.timer


+==================================================================================================+
|  SECTION 08  //  TROUBLESHOOT  —  one fix per symptom                                            |
+==================================================================================================+

    SYMPTOM: "could not connect to ollama server"
    FIX:     systemctl --user start ollama.service
             sleep 3
             curl -fsS http://127.0.0.1:11434/api/tags

    SYMPTOM: Service failed — port 11434 in use
    FIX:     ss -tlnp | grep 11434              find the squatter
             pkill -f "ollama serve"            kill stray host ollama
             systemctl --user restart ollama.service

    SYMPTOM: Wrong personality / wrong hardware info
    FIX:     cd ~/projects/cheatsheets
             ollama create avalhla -f persona/avalhla.Modelfile

    SYMPTOM: Slow generation
    FIX:     Close Firefox, OBS, games. Check nvidia-smi.

    SYMPTOM: OOM / GPU spillover
    FIX:     avastop. Close apps. avastart.
             Or lower num_ctx to 8192 in Modelfile and rebuild.

    SYMPTOM: Nuclear reset (model corrupted)
    FIX:     ollama rm avalhla
             ollama create avalhla -f persona/avalhla.Modelfile

+==================================================================================================+
|  SECTION 10  //  SYSTEM UPDATES  —  YOU run these. Never topgrade.                                |
+==================================================================================================+

    ujust update                Bazzite one-shot update (OS + flatpak)
    rpm-ostree status           current + previous deployment
    sudo rpm-ostree rollback    boot previous working
    systemctl reboot            apply pending deployment

    RULES:
      topgrade must NOT run ujust   (check ~/.config/topgrade.toml)
      fstrim runs on its own timer  (never inside topgrade)
      never sudo dnf install on host
