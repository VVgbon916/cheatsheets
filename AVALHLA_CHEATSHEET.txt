+==================================================================================================+
|  SECTION 01  //  START HER UP  //  one systemd command                                          |
+==================================================================================================+

    # Start the Ollama container + open chat
    ~/Avalhla/repo/scripts/start-avalhla.sh

    # Or manually:
    systemctl --user start ollama.service
    ollama run avalhla

    # The container runs docker.io/ollama/ollama as a systemd user service.
    # No Distrobox. No pkill. No env vars at start time.


+==================================================================================================+
|  SECTION 02  //  STOP HER  //  free the VRAM                                                    |
+==================================================================================================+

    # Preferred (frees ~5.5 GB VRAM)
    systemctl --user stop ollama.service

    # Verify stopped
    systemctl --user status ollama.service --no-pager | head -5
    # Expect: Active: inactive (dead)


+==================================================================================================+
|  SECTION 03  //  TALK TO HER                                                                    |
+==================================================================================================+

    # Interactive chat
    ollama run avalhla

    # One-shot question
    ollama run avalhla "your question here"

    # Inside the >>> prompt:
    /bye            exit cleanly
    Ctrl+D          same as /bye
    /?              help
    /clear          wipe current chat context
    /show info      current model details


+==================================================================================================+
|  SECTION 04  //  MEMORY TOOLS  //  persist across sessions                                      |
+==================================================================================================+

    ai-chat                 persistent chat, saves to ~/.ai-memory/
                            /bye exit  /clear reset  /history last 10  /mem profile

    ai-learn <path>         index a file or folder into her knowledge base

    ai-remember             she reflects on recent sessions
                            saves summary to ~/.ai-memory/reflections/

    ai-ask "question"       one-shot with recent context (uses last N turns)
                            ai-ask -n 10 "summarize our last chat"


+==================================================================================================+
|  SECTION 05  //  EDIT + REBUILD  //  only after editing persona/avalhla.Modelfile              |
+==================================================================================================+

    # Edit the Modelfile
    nano ~/Avalhla/repo/persona/avalhla.Modelfile

    # Rebuild (server must be running)
    cd ~/Avalhla/repo
    ollama create avalhla -f persona/avalhla.Modelfile

    # Relaunch
    ollama run avalhla


+==================================================================================================+
|  SECTION 06  //  INSPECT  //  what is she right now                                             |
+==================================================================================================+

    ollama list | grep avalhla                  # does she exist?
    ollama show avalhla --modelfile             # full build
    ollama show avalhla --parameters            # temperature, top_k, ctx
    ollama show avalhla --system                # just the SYSTEM block

    # Expected parameters:
    #   temperature     0.3
    #   top_p           0.9
    #   top_k           40
    #   repeat_penalty  1.05
    #   num_predict     2048
    #   num_ctx         16384


+==================================================================================================+
|  SECTION 07  //  TROUBLESHOOT                                                                   |
+==================================================================================================+

    SYMPTOM: "could not connect to ollama server"
    FIX:     Start the container.
             systemctl --user start ollama.service
             sleep 3
             curl -fsS http://127.0.0.1:11434/api/tags

    SYMPTOM: Service failed — port in use
    FIX:     Something else is on 11434.
             ss -tlnp | grep 11434
             pkill -f "ollama serve"        # kill stray host ollama
             systemctl --user restart ollama.service

    SYMPTOM: Wrong personality
    FIX:     cd ~/Avalhla/repo
             ollama create avalhla -f persona/avalhla.Modelfile

    SYMPTOM: Slow generation
    FIX:     Close Firefox, OBS, games. Check nvidia-smi for VRAM use.

    SYMPTOM: Nuclear reset (model corrupted)
    FIX:     ollama rm avalhla
             ollama create avalhla -f persona/avalhla.Modelfile


+==================================================================================================+
|  SECTION 08  //  SYSTEM UPDATES  //  YOU run these. Never topgrade.                            |
+==================================================================================================+

    ujust update                # Bazzite one-shot update (OS + flatpak)
    rpm-ostree status           # current + previous deployment
    sudo rpm-ostree rollback    # boot previous working
    systemctl reboot            # apply pending deployment

    # The Ollama container auto-updates at midnight via:
    #   podman-auto-update.timer  (user service)

    RULES:
      topgrade must NOT run ujust
      fstrim runs on its own timer
      never sudo dnf install on host
